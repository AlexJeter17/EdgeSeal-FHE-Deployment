% Example LaTeX code for plotting EdgeSeal-FHE data with pgfplots
% Requires: \usepackage{pgfplots} \pgfplotsset{compat=1.18}

% ==============================================================================
% EXAMPLE 1: Encryption Time vs Security Level (Log Scale)
% ==============================================================================
\begin{figure}[!t]
\centering
\begin{tikzpicture}
\begin{semilogyaxis}[
    xlabel={Polynomial Degree},
    ylabel={Encryption Time (ms, log scale)},
    legend style={at={(0.5,-0.15)}, anchor=north, legend columns=3},
    width=0.8\textwidth,
    height=6cm,
    grid=major,
    xtick={4096,8192,16384},
    xticklabels={4096\\(80-bit),8192\\(128-bit),16384\\(256-bit)}
]
% EdgeSeal-FHE
\addplot[blue,mark=o,thick] table[x=PolyDegree, y=EdgeSeal_ms, col sep=comma] {encryption_time_vs_security.csv};
% TenSEAL/SEAL Baseline
\addplot[purple,mark=square,thick,dashed] table[x=PolyDegree, y=TenSEAL_ms, col sep=comma] {encryption_time_vs_security.csv};
% AES-256 Reference
\addplot[red,dotted,thick] table[x=PolyDegree, y=AES256_ms, col sep=comma] {encryption_time_vs_security.csv};
\legend{EdgeSeal-FHE, TenSEAL/SEAL Baseline, AES-256 Reference}
\end{semilogyaxis}
\end{tikzpicture}
\caption{Encryption Time vs Security Level (CKKS Polynomial Degree)}
\label{fig:encryption-time}
\end{figure}

% ==============================================================================
% EXAMPLE 3: Embedding Quality Preservation (Bar Chart)
% ==============================================================================
\begin{figure}[!t]
\centering
\begin{tikzpicture}
\begin{axis}[
    ybar,
    bar width=0.3cm,
    xlabel={Configuration},
    ylabel={Reconstruction Quality (\%)},
    symbolic x coords={Poly4096_80bit,Poly8192_128bit,Poly16384_256bit,Average},
    xtick=data,
    legend pos=north east,
    width=0.8\textwidth,
    height=6cm,
    ymin=99,
    ymax=100.25,
    grid=major,
    nodes near coords,
    every node near coord/.append style={font=\tiny}
]
\addplot[fill=blue!50] table[x=Configuration, y=Plaintext_Quality, col sep=comma] {embedding_quality_preservation.csv};
\addplot[fill=purple!50] table[x=Configuration, y=FHE_Quality, col sep=comma] {embedding_quality_preservation.csv};
\legend{Plaintext, FHE Encrypted}
\end{axis}
\end{tikzpicture}
\caption{Embedding Quality Preservation: Plaintext vs FHE}
\label{fig:quality-preservation}
\end{figure}
